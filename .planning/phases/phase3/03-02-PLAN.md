    /**
     * Check if webhook was already processed (idempotency).
     * Returns true if already processed (should skip).
     */
    export async function checkIdempotency(transactionId: string): Promise<boolean> {
      const existing = await prisma.processedWebhook.findUnique({
        where: { externalId: transactionId },
      });
      return !!existing;
    }

    /**
     * Record webhook as processed (for idempotency).
     */
    export async function recordWebhookProcessed(
      transactionId: string,
      eventType: string,
      payload: any
    ) {
      await prisma.processedWebhook.create({
        data: {
          id: nanoid(),
          externalId: transactionId,
          eventType,
          payload,
        },
      });
    }

    import { NextRequest, NextResponse } from 'next/server';
    import { prisma } from '@/core/database/client';
    import { nanoid } from 'nanoid';
    import {
      verifySelcomSignature,
      parseSelcomWebhookPayload,
      checkIdempotency,
      recordWebhookProcessed,
    } from '@/integrations/selcom/webhooks';
    import { settlePayment } from '@/features/wallets/payment-saga';
    import { updateInvoiceStatus } from '@/features/invoices/actions';

    export async function POST(request: NextRequest) {
      try {
        // Read raw body for signature verification
        const body = await request.json();
        const signature = request.headers.get('x-selcom-signature');
        const timestamp = request.headers.get('x-selcom-timestamp');

        if (!signature || !timestamp) {
          return NextResponse.json(
            { error: 'Missing signature headers' },
            { status: 400 }
          );
        }

        // Parse and validate payload
        const payload = parseSelcomWebhookPayload(body);
        if (!payload) {
          return NextResponse.json(
            { error: 'Invalid payload' },
            { status: 400 }
          );
        }

        // Add signature to payload for verification
        const payloadWithSig = { ...payload, signature };

        // Verify HMAC signature
        const isValid = verifySelcomSignature(payloadWithSig);
        if (!isValid) {
          console.warn('[Selcom Webhook] Invalid signature', {
            transactionId: payload.transactionId,
          });
          return NextResponse.json(
            { error: 'Invalid signature' },
            { status: 401 }
          );
        }

        // Check idempotency (duplicate detection)
        const alreadyProcessed = await checkIdempotency(payload.transactionId);
        if (alreadyProcessed) {
          console.log('[Selcom Webhook] Duplicate detected, returning 200', {
            transactionId: payload.transactionId,
          });
          return NextResponse.json({ status: 'already_processed' });
        }

        // Record webhook as processed (idempotency)
        await recordWebhookProcessed(
          payload.transactionId,
          payload.status,
          payload
        );

        // Process based on status
        if (payload.status === 'SUCCESS') {
          await settlePayment(payload.invoiceId, payload.transactionId);
          await updateInvoiceStatus({
            invoiceId: payload.invoiceId,
            status: 'PAID',
          });
        } else if (payload.status === 'FAILED') {
          await updateInvoiceStatus({
            invoiceId: payload.invoiceId,
            status: 'FAILED',
          });
          // Trigger retry logic or suspension
          await handleFailedPayment(payload.invoiceId);
        }

        return NextResponse.json({ status: 'processed' });
      } catch (error) {
        console.error('[Selcom Webhook] Error processing webhook:', error);
        return NextResponse.json(
          { error: 'Internal server error' },
          { status: 500 }
        );
      }
    }

    /**
     * Handle failed payment: increment retry count, suspend if max retries exceeded
     */
    async function handleFailedPayment(invoiceId: string) {
      const invoice = await prisma.invoice.findUnique({
        where: { id: invoiceId },
        include: { assignment: true },
      });

      if (!invoice) return;

      const newRetryCount = invoice.retryCount + 1;

      if (newRetryCount >= invoice.maxRetries) {
        // Max retries exceeded - suspend agreement
        await prisma.invoice.update({
          where: { id: invoiceId },
          data: {
            retryCount: newRetryCount,
            status: 'CANCELLED',
          },
        });

        // Suspend agreement
        await prisma.assignment.update({
          where: { id: invoice.assignmentId },
          data: { status: 'CANCELLED_NO_SHOW' },
        });
      } else {
        // Increment retry count, keep PENDING
        await prisma.invoice.update({
          where: { id: invoiceId },
          data: { retryCount: newRetryCount },
        });
      }
    }

    'use server';

    import { prisma } from '@/core/database/client';
    import { selcom } from '@/integrations/selcom/client';
    import { nanoid } from 'nanoid';
    import { Decimal } from '@prisma/client/runtime/library';
    import { Prisma } from '@prisma/client';

    export interface PaymentSagaData {
      assignmentId: string;
      invoiceId: string;
      totalAmount: number;
      ownerPhone: string;
      providerPhone: string;
    }

    /**
     * Execute payment saga with 80/20 split.
     * Atomic transaction: if any step fails, all changes are rolled back.
     */
    export async function executePaymentSaga(
      data: PaymentSagaData
    ): Promise<{
      success: boolean;
      transactionId?: string;
      error?: string;
    }> {
      const sagaId = nanoid();
      console.log(
        `[PAYMENT SAGA] Starting saga ${sagaId} for assignment ${data.assignmentId}`
      );

      try {
        return await prisma.$transaction(
          async (tx) => {
            // Step 1: Request payment from owner via Selcom
            const paymentResult = await selcom.requestPayment({
              invoiceId: data.invoiceId,
              amount: data.totalAmount,
              phoneNumber: data.ownerPhone,
              description: `Service payment for assignment ${data.assignmentId}`,
            });

            if (!paymentResult.success) {
              throw new Error(
                `Owner payment failed: ${paymentResult.error}`
              );
            }

            // Step 2: Calculate split
            const total = new Decimal(data.totalAmount);
            const providerAmount = total.mul(0.8);
            const platformFee = total.mul(0.2);

            // Step 3: Get or create provider wallet
            const assignment = await tx.assignment.findUnique({
              where: { id: data.assignmentId },
              select: { providerId: true },
            });

            if (!assignment?.providerId) {
              throw new Error('Assignment has no provider');
            }

            let wallet = await tx.wallet.findUnique({
              where: { providerId: assignment.providerId },
            });

            if (!wallet) {
              wallet = await tx.wallet.create({
                data: {
                  id: nanoid(),
                  providerId: assignment.providerId,
                  availableBalance: new Decimal(0),
                  pendingBalance: new Decimal(0),
                },
              });
            }

            // Step 4: Credit provider wallet (pending until webhook confirms)
            await tx.wallet.update({
              where: { id: wallet.id },
              data: {
                pendingBalance: { increment: providerAmount },
              },
            });

            // Step 5: Record provider credit transaction
            await tx.walletTransaction.create({
              data: {
                id: nanoid(),
                walletId: wallet.id,
                type: 'CREDIT',
                amount: providerAmount,
                reference: data.assignmentId,
                status: 'PENDING', // Will be SETTLED when webhook confirms
              },
            });

            // Step 6: Record platform fee (audit only, no wallet)
            await tx.financialAuditLog.create({
              data: {
                id: nanoid(),
                entityType: 'ASSIGNMENT',
                entityId: data.assignmentId,
                action: 'PAYMENT_SPLIT',
                payload: {
                  sagaId,
                  totalAmount: total.toString(),
                  providerAmount: providerAmount.toString(),
                  platformFee: platformFee.toString(),
                  status: 'PENDING_SETTLEMENT',
                },
              },
            });

            console.log(
              `[PAYMENT SAGA] Completed successfully ${sagaId}`
            );
            return {
              success: true,
              transactionId: paymentResult.transactionId,
            };
          },
          { isolationLevel: Prisma.TransactionIsolationLevel.Serializable }
        );
      } catch (error) {
        console.error(`[PAYMENT SAGA] Failed ${sagaId}:`, error);

        // Record failure in audit log
        await prisma.financialAuditLog.create({
          data: {
            id: nanoid(),
            entityType: 'ASSIGNMENT',
            entityId: data.assignmentId,
            action: 'SAGA_FAILED',
            payload: {
              sagaId,
              error: error instanceof Error ? error.message : 'Unknown error',
            },
          },
        });

        return {
          success: false,
          error: error instanceof Error ? error.message : 'Payment failed',
        };
      }
    }

    /**
     * Settle payment when Selcom webhook confirms.
     * Moves transaction from PENDING to SETTLED, updates wallet balances.
     */
    export async function settlePayment(
      invoiceId: string,
      transactionId: string
    ): Promise<void> {
      await prisma.$transaction(
        async (tx) => {
          // Find pending transaction for this assignment
          const invoice = await tx.invoice.findUnique({
            where: { id: invoiceId },
            include: { assignment: true },
          });

          if (!invoice) throw new Error('Invoice not found');

          // Update wallet transaction to SETTLED
          const transaction = await tx.walletTransaction.findFirst({
            where: {
              reference: invoice.assignmentId,
              status: 'PENDING',
            },
          });

          if (!transaction) throw new Error('Pending transaction not found');

          await tx.walletTransaction.update({
            where: { id: transaction.id },
            data: { status: 'SETTLED' },
          });

          // Move from pending to available balance
          const wallet = await tx.wallet.findUnique({
            where: { id: transaction.walletId },
          });

          if (!wallet) throw new Error('Wallet not found');

          const newPending = wallet.pendingBalance.minus(transaction.amount);
          const newAvailable = wallet.availableBalance.plus(transaction.amount);

          await tx.wallet.update({
            where: { id: wallet.id },
            data: {
              pendingBalance: newPending,
              availableBalance: newAvailable,
            },
          });

          // Update audit log
          await tx.financialAuditLog.create({
            data: {
              id: nanoid(),
              entityType: 'ASSIGNMENT',
              entityId: invoice.assignmentId,
              action: 'PAYMENT_SETTLED',
              payload: {
                transactionId,
                amount: transaction.amount.toString(),
                walletId: wallet.id,
              },
            },
          });
        },
        { isolationLevel: Prisma.TransactionIsolationLevel.Serializable }
      );
    }
