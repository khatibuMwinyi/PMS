    ```typescript
    const MIN_WITHDRAWAL_AMOUNT = 10000; // TZS 10,000 minimum

    /**
     * Request withdrawal from wallet.
     * Checks minimum threshold and available balance.
     */
    export async function requestWithdrawal(input: WithdrawalRequestInput) {
      const session = await auth();
      if (!session?.user || session.user.role !== 'PROVIDER') {
        throw new Error('Unauthorized: Provider only');
      }

      const parsed = WithdrawalRequestSchema.safeParse(input);
      if (!parsed.success) {
        throw new Error(parsed.error.errors[0]?.message ?? 'Invalid input');
      }

      const { amount, providerPhone } = parsed.data;

      // Check minimum threshold
      if (amount < MIN_WITHDRAWAL_AMOUNT) {
        throw new Error(`Minimum withdrawal amount is ${MIN_WITHDRAWAL_AMOUNT} TZS`);
      }

      // Get provider wallet
      const provider = await prisma.providerProfile.findUnique({
        where: { userId: session.user.id },
        include: { wallet: true },
      });

      if (!provider?.wallet) {
        throw new Error('Wallet not found');
      }

      // Check available balance
      if (provider.wallet.availableBalance.toNumber() < amount) {
        throw new Error('Insufficient available balance');
      }

      // Create withdrawal request (pending review)
      const withdrawal = await prisma.withdrawalRequest.create({
        data: {
          id: nanoid(),
          walletId: provider.wallet.id,
          amount: new Decimal(amount),
          providerPhone,
          status: 'PENDING_REVIEW',
        },
      });

      // Temporarily reduce available balance (hold funds)
      await prisma.wallet.update({
        where: { id: provider.wallet.id },
        data: {
          availableBalance: { decrement: amount },
          pendingBalance: { increment: amount },
        },
      });

      return withdrawal;
    }

    /**
     * Admin/Staff: Approve withdrawal request.
     * Triggers Selcom disbursement.
     */
    export async function approveWithdrawal(withdrawalId: string) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      const withdrawal = await prisma.withdrawalRequest.findUnique({
        where: { id: withdrawalId },
        include: { wallet: true },
      });

      if (!withdrawal) throw new Error('Withdrawal request not found');
      if (withdrawal.status !== 'PENDING_REVIEW') {
        throw new Error('Withdrawal not pending review');
      }

      // Update status to APPROVED
      await prisma.withdrawalRequest.update({
        where: { id: withdrawalId },
        data: {
          status: 'APPROVED',
          reviewedAt: new Date(),
        },
      });

      // Trigger Selcom disbursement
      const selcom = new SelcomClient();
      const result = await selcom.disburseFunds(
        withdrawal.providerPhone,
        withdrawal.amount.toNumber(),
        withdrawal.id
      );

      if (result.success) {
        // Update to PROCESSING
        await prisma.withdrawalRequest.update({
          where: { id: withdrawalId },
          data: { status: 'PROCESSING' },
        });

        // Record audit
        await prisma.financialAuditLog.create({
          data: {
            id: nanoid(),
            entityType: 'WITHDRAWAL',
            entityId: withdrawalId,
            action: 'WITHDRAWAL_APPROVED',
            payload: {
              amount: withdrawal.amount.toString(),
              providerPhone: withdrawal.providerPhone,
              transactionId: result.transactionId,
            },
          },
        });
      } else {
        // Revert to PENDING_REVIEW on failure
        await prisma.withdrawalRequest.update({
          where: { id: withdrawalId },
          data: { status: 'PENDING_REVIEW' },
        });
        throw new Error(`Disbursement failed: ${result.error}`);
      }

      return withdrawal;
    }

    /**
     * Admin/Staff: Reject withdrawal request.
     * Returns funds to available balance.
     */
    export async function rejectWithdrawal(withdrawalId: string, reason: string) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      const withdrawal = await prisma.withdrawalRequest.findUnique({
        where: { id: withdrawalId },
        include: { wallet: true },
      });

      if (!withdrawal) throw new Error('Withdrawal request not found');

      // Update status to REJECTED
      await prisma.withdrawalRequest.update({
        where: { id: withdrawalId },
        data: {
          status: 'REJECTED',
          reviewedAt: new Date(),
        },
      });

      // Return funds to available balance
      await prisma.wallet.update({
        where: { id: withdrawal.walletId },
        data: {
          availableBalance: { increment: withdrawal.amount },
          pendingBalance: { decrement: withdrawal.amount },
        },
      });

      return withdrawal;
    }

    /**
     * Get withdrawal requests (admin/staff see all, provider sees own).
     */
    export async function getWithdrawalRequests(walletId?: string) {
      const session = await auth();
      if (!session?.user) throw new Error('Unauthorized');

      if (['ADMIN', 'STAFF'].includes(session.user.role)) {
        // Admin/staff can see all or filter by walletId
        return await prisma.withdrawalRequest.findMany({
          where: walletId ? { walletId } : {},
          include: {
            wallet: { include: { provider: true } },
          },
          orderBy: { createdAt: 'desc' },
        });
      } else {
        // Provider sees only their withdrawals
        const provider = await prisma.providerProfile.findUnique({
          where: { userId: session.user.id },
          include: { wallet: true },
        });

        if (!provider?.wallet) return [];

        return await prisma.withdrawalRequest.findMany({
          where: { walletId: provider.wallet.id },
          orderBy: { createdAt: 'desc' },
        });
      }
    }
    ```

    ```typescript
    import { auth } from '@/core/auth';

    export async function POST(request: NextRequest) {
      const session = await auth();
      if (!session?.user || session.user.role !== 'PROVIDER') {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
      }

      try {
        const body = await request.json();
        const result = await requestWithdrawal(body);
        return NextResponse.json(result);
      } catch (error) {
        return NextResponse.json(
          { error: error instanceof Error ? error.message : 'Request failed' },
          { status: 400 }
        );
      }
    }

    export async function GET(request: NextRequest) {
      const session = await auth();
      if (!session?.user) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
      }

      try {
        const { getWithdrawalRequests } = await import('@/features/wallets/actions');
        const withdrawals = await getWithdrawalRequests();
        return NextResponse.json(withdrawals);
      } catch (error) {
        return NextResponse.json(
          { error: 'Failed to fetch withdrawals' },
          { status: 500 }
        );
      }
    }
    ```

    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { auth } from '@/core/auth';
    import { approveWithdrawal, rejectWithdrawal } from '@/features/wallets/actions';

    export async function POST(
      request: NextRequest,
      { params }: { params: { id: string } }
    ) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
      }

      try {
        const body = await request.json();
        const { action, reason } = body;

        if (action === 'APPROVE') {
          await approveWithdrawal(params.id);
        } else if (action === 'REJECT') {
          await rejectWithdrawal(params.id, reason);
        } else {
          return NextResponse.json({ error: 'Invalid action' }, { status: 400 });
        }

        return NextResponse.json({ success: true });
      } catch (error) {
        return NextResponse.json(
          { error: error instanceof Error ? error.message : 'Review failed' },
          { status: 400 }
        );
      }
    }
    ```

    ## src/app/api/webhooks/selcom/disburse/route.ts (Disbursement webhook)
    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { prisma } from '@/core/database/client';
    import { nanoid } from 'nanoid';
    import { verifySelcomSignature } from '@/integrations/selcom/webhooks';

    export async function POST(request: NextRequest) {
      try {
        const body = await request.json();
        const signature = request.headers.get('x-selcom-signature');

        if (!signature) {
          return NextResponse.json({ error: 'Missing signature' }, { status: 400 });
        }

        // Verify signature
        const isValid = verifySelcomSignature({ ...body, signature });
        if (!isValid) {
          return NextResponse.json({ error: 'Invalid signature' }, { status: 401 });
        }

        const { transactionId, withdrawalId, status } = body;

        // Check idempotency
        const existing = await prisma.processedWebhook.findUnique({
          where: { externalId: transactionId },
        });

        if (existing) {
          return NextResponse.json({ status: 'already_processed' });
        }

        // Record webhook
        await prisma.processedWebhook.create({
          data: {
            id: nanoid(),
            externalId: transactionId,
            eventType: 'DISBURSEMENT',
            payload: body,
          },
        });

        // Update withdrawal status
        if (status === 'SUCCESS') {
          await prisma.withdrawalRequest.update({
            where: { id: withdrawalId },
            data: {
              status: 'COMPLETED',
              processedAt: new Date(),
            },
          });

          // Update wallet: remove from pending
          const withdrawal = await prisma.withdrawalRequest.findUnique({
            where: { id: withdrawalId },
          });

          if (withdrawal) {
            await prisma.wallet.update({
              where: { id: withdrawal.walletId },
              data: {
                pendingBalance: { decrement: withdrawal.amount },
              },
            });
          }
        } else if (status === 'FAILED') {
          await prisma.withdrawalRequest.update({
            where: { id: withdrawalId },
            data: {
              status: 'PENDING_REVIEW', // Back to review
              processedAt: new Date(),
            },
          });

          // Return funds to available
          const withdrawal = await prisma.withdrawalRequest.findUnique({
            where: { id: withdrawalId },
          });

          if (withdrawal) {
            await prisma.wallet.update({
              where: { id: withdrawal.walletId },
              data: {
                availableBalance: { increment: withdrawal.amount },
                pendingBalance: { decrement: withdrawal.amount },
              },
            });
          }
        }

        return NextResponse.json({ status: 'processed' });
      } catch (error) {
        console.error('[Disbursement Webhook] Error:', error);
        return NextResponse.json(
          { error: 'Internal server error' },
          { status: 500 }
        );
      }
    }
    ```

    ```typescript
    /**
     * Get wallet summary with total earned (sum of all settled CREDIT transactions).
     */
    export async function getProviderWallet(): Promise<WalletSummary | null> {
      const session = await auth();
      if (!session?.user || session.user.role !== 'PROVIDER') {
        throw new Error('Unauthorized');
      }

      const provider = await prisma.providerProfile.findUnique({
        where: { userId: session.user.id },
        select: { id: true },
      });

      if (!provider) return null;

      const wallet = await prisma.wallet.findUnique({
        where: { providerId: provider.id },
        include: {
          transactions: {
            orderBy: { createdAt: 'desc' },
            take: 50,
          },
        },
      });

      if (!wallet) return null;

      // Calculate total earned (sum of all settled CREDIT transactions)
      const totalEarnedResult = await prisma.walletTransaction.aggregate({
        where: {
          walletId: wallet.id,
          type: 'CREDIT',
          status: 'SETTLED',
        },
        _sum: { amount: true },
      });

      const totalEarned = totalEarnedResult._sum.amount?.toNumber() ?? 0;

      return {
        id: wallet.id,
        availableBalance: wallet.availableBalance.toNumber(),
        pendingBalance: wallet.pendingBalance.toNumber(),
        totalEarned,
        lastUpdated: wallet.updatedAt,
        transactions: wallet.transactions.map((t) => ({
          id: t.id,
          type: t.type,
          amount: t.amount.toNumber(),
          status: t.status,
          createdAt: t.createdAt.toISOString(),
        })),
      };
    }

    /**
     * Get immutable ledger (all transactions, never updated/deleted).
     * Used for audit and provider statement.
     */
    export async function getWalletLedger(walletId?: string) {
      const session = await auth();
      if (!session?.user) throw new Error('Unauthorized');

      let targetWalletId = walletId;

      if (!targetWalletId) {
        // Get own wallet
        const provider = await prisma.providerProfile.findUnique({
          where: { userId: session.user.id },
          include: { wallet: true },
        });
        targetWalletId = provider?.wallet?.id;
      }

      if (!targetWalletId) return [];

      return await prisma.walletTransaction.findMany({
        where: { walletId: targetWalletId },
        orderBy: { createdAt: 'asc' }, // Chronological order for ledger
      });
    }

    /**
     * Admin: Get all wallets with total earned for reconciliation.
     */
    export async function getAllWallets() {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      const wallets = await prisma.wallet.findMany({
        include: {
          provider: { select: { businessName: true, userId: true } },
          alerts: { where: { resolvedAt: null }, select: { id: true } },
          transactions: {
            where: { type: 'CREDIT', status: 'SETTLED' },
            select: { amount: true },
          },
        },
        orderBy: { availableBalance: 'desc' },
      });

      // Calculate total earned per wallet
      return wallets.map((wallet) => ({
        ...wallet,
        totalEarned: wallet.transactions.reduce(
          (sum, t) => sum + t.amount.toNumber(),
          0
        ),
      }));
    }
    ```
