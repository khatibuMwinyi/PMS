    ```typescript
    'use server';
    import { prisma } from '@/core/database/client';
    import { nanoid } from 'nanoid';
    import { auth } from '@/core/auth';
    import { SuspendAgreementSchema, TerminateAgreementSchema } from './types';

    /**
     * Suspend agreement due to payment failure.
     * Sets status to SUSPENDED, records suspension period.
     */
    export async function suspendAgreement(input: SuspendAgreementInput) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      const parsed = SuspendAgreementSchema.safeParse(input);
      if (!parsed.success) {
        throw new Error(parsed.error.errors[0]?.message ?? 'Invalid input');
      }

      const { agreementId, reason, suspendDays } = parsed.data;

      const agreement = await prisma.assignment.findUnique({
        where: { id: agreementId },
        include: { invoices: true },
      });

      if (!agreement) throw new Error('Agreement not found');

      // Calculate suspension end date
      const suspendedUntil = new Date();
      suspendedUntil.setDate(suspendedUntil.getDate() + suspendDays);

      // Update assignment status to SUSPENDED
      await prisma.assignment.update({
        where: { id: agreementId },
        data: {
          status: 'SUSPENDED',
          // Note: Add suspendedUntil field to schema if needed
        },
      });

      // Record audit
      await prisma.financialAuditLog.create({
        data: {
          id: nanoid(),
          entityType: 'ASSIGNMENT',
          entityId: agreementId,
          action: 'AGREEMENT_SUSPENDED',
          payload: {
            reason,
            suspendedUntil: suspendedUntil.toISOString(),
            invoiceRetryCount: agreement.invoices?.[0]?.retryCount ?? 0,
          },
        },
      });

      // Send suspension notification
      const { sendSuspensionAlert } = await import('@/features/notifications/actions');
      await sendSuspensionAlert(agreementId, reason);

      return { success: true, suspendedUntil };
    }

    /**
     * Terminate agreement permanently.
     * Used for persistent payment failures or policy violations.
     */
    export async function terminateAgreement(input: TerminateAgreementInput) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      const parsed = TerminateAgreementSchema.safeParse(input);
      if (!parsed.success) {
        throw new Error(parsed.error.errors[0]?.message ?? 'Invalid input');
      }

      const { agreementId, reason } = parsed.data;

      const agreement = await prisma.assignment.findUnique({
        where: { id: agreementId },
      });

      if (!agreement) throw new Error('Agreement not found');

      // Update status to CANCELLED (or add TERMINATED status)
      await prisma.assignment.update({
        where: { id: agreementId },
        data: { status: 'CANCELLED_NO_SHOW' }, // Reuse existing terminated state
      });

      // Record audit
      await prisma.financialAuditLog.create({
        data: {
          id: nanoid(),
          entityType: 'ASSIGNMENT',
          entityId: agreementId,
          action: 'AGREEMENT_TERMINATED',
          payload: { reason },
        },
      });

      // Send termination notification
      const { sendTerminationNotice } = await import('@/features/notifications/actions');
      await sendTerminationNotice(agreementId, reason);

      return { success: true };
    }

    /**
     * Auto-suspend agreement when invoice max retries exceeded.
     * Called by background job or webhook handler.
     */
    export async function checkAndSuspendOnPaymentFailure(invoiceId: string) {
      const invoice = await prisma.invoice.findUnique({
        where: { id: invoiceId },
        include: { assignment: true },
      });

      if (!invoice) throw new Error('Invoice not found');

      if (invoice.status === 'FAILED' && invoice.retryCount >= invoice.maxRetries) {
        await suspendAgreement({
          agreementId: invoice.assignmentId,
          reason: `Payment failed after ${invoice.maxRetries} retries`,
          suspendDays: 30,
        });
      }
    }
    ```

    ```typescript
    export async function sendPaymentSuccessNotification(
      invoiceId: string,
      transactionId: string
    ) {
      const invoice = await prisma.invoice.findUnique({
        where: { id: invoiceId },
        include: {
          assignment: {
            include: {
              property: { include: { owner: true } },
              provider: { include: { user: true } },
            },
          },
        },
      });

      if (!invoice) throw new Error('Invoice not found');

      const ownerPhone = invoice.assignment.property.owner.phone;
      const providerPhone = invoice.assignment.provider?.user?.phone;

      const message = `Payment of ${invoice.amount} TZS successful. Transaction: ${transactionId}. Thank you!`;

      // Send to owner
      await sendSMS(ownerPhone, message);
      await prisma.notification.create({
        data: {
          id: nanoid(),
          userId: invoice.assignment.property.ownerId,
          type: 'SMS',
          event: 'PAYMENT_SUCCESS',
          payload: { invoiceId, transactionId, amount: invoice.amount },
        },
      });

      // Send to provider (if phone available)
      if (providerPhone) {
        const providerMessage = `Payment of ${invoice.amount * 0.8} TZS credited to your wallet. Transaction: ${transactionId}`;
        await sendSMS(providerPhone, providerMessage);
      }
    }

    /**
     * Send payment failure reminder with retry information.
     */
    export async function sendPaymentFailureNotification(
      invoiceId: string,
      willRetry: boolean
    ) {
      const invoice = await prisma.invoice.findUnique({
        where: { id: invoiceId },
        include: {
          assignment: {
            include: {
              property: { include: { owner: true } },
            },
          },
        },
      });

      if (!invoice) throw new Error('Invoice not found');

      const ownerPhone = invoice.assignment.property.owner.phone;
      const retryInfo = willRetry
        ? `Retry attempt ${invoice.retryCount + 1}/${invoice.maxRetries}.`
        : `Max retries (${invoice.maxRetries}) exceeded. Agreement may be suspended.`;

      const message = `Payment of ${invoice.amount} TZS failed. ${retryInfo}`;

      await sendSMS(ownerPhone, message);
      await prisma.notification.create({
        data: {
          id: nanoid(),
          userId: invoice.assignment.property.ownerId,
          type: 'SMS',
          event: 'PAYMENT_FAILED',
          payload: {
            invoiceId,
            retryCount: invoice.retryCount,
            maxRetries: invoice.maxRetries,
            willRetry,
          },
        },
      });
    }

    /**
     * Send suspension alert to owner and provider.
     */
    export async function sendSuspensionAlert(agreementId: string, reason: string) {
      const assignment = await prisma.assignment.findUnique({
        where: { id: agreementId },
        include: {
          property: { include: { owner: true } },
          provider: { include: { user: true } },
        },
      });

      if (!assignment) throw new Error('Assignment not found');

      const ownerPhone = assignment.property.owner.phone;
      const providerPhone = assignment.provider?.user?.phone;

      const message = `Agreement suspended. Reason: ${reason}. Contact support.`;

      // Notify owner
      await sendSMS(ownerPhone, message);
      await prisma.notification.create({
        data: {
          id: nanoid(),
          userId: assignment.property.ownerId,
          type: 'SMS',
          event: 'AGREEMENT_SUSPENDED',
          payload: { agreementId, reason },
        },
      });

      // Notify provider
      if (providerPhone) {
        await sendSMS(providerPhone, message);
      }
    }

    /**
     * Send withdrawal status update to provider.
     */
    export async function sendWithdrawalStatusUpdate(
      withdrawalId: string,
      status: 'APPROVED' | 'REJECTED' | 'COMPLETED'
    ) {
      const withdrawal = await prisma.withdrawalRequest.findUnique({
        where: { id: withdrawalId },
        include: {
          wallet: { include: { provider: { include: { user: true } } } },
        },
      });

      if (!withdrawal) throw new Error('Withdrawal not found');

      const providerPhone = withdrawal.wallet.provider?.user?.phone;
      if (!providerPhone) return;

      const messages = {
        APPROVED: `Withdrawal of ${withdrawal.amount} TZS approved. Processing...`,
        REJECTED: `Withdrawal of ${withdrawal.amount} TZS rejected. Contact support.`,
        COMPLETED: `Withdrawal of ${withdrawal.amount} TZS completed. Check your mobile money.`,
      };

      await sendSMS(providerPhone, messages[status]);
      await prisma.notification.create({
        data: {
          id: nanoid(),
          userId: withdrawal.wallet.provider!.userId,
          type: 'SMS',
          event: 'WITHDRAWAL_STATUS',
          payload: { withdrawalId, status, amount: withdrawal.amount },
        },
      });
    }
    ```

    ## Create src/core/jobs/workers/payment-monitor.ts

    ```typescript
    import { prisma } from '@/core/database/client';
    import { checkAndSuspendOnPaymentFailure } from '@/features/agreements/actions';
    import { sendPaymentFailureNotification } from '@/features/notifications/actions';

    export const name = 'payment-monitor';

    export async function handler() {
      console.log('[Payment Monitor] Checking for failed payments...');

      // Find invoices that are FAILED and need action
      const failedInvoices = await prisma.invoice.findMany({
        where: {
          status: 'FAILED',
          retryCount: { gt: 0 },
        },
        include: { assignment: true },
      });

      for (const invoice of failedInvoices) {
        try {
          const willRetry = invoice.retryCount < invoice.maxRetries;

          if (!willRetry) {
            // Max retries exceeded - suspend agreement
            await checkAndSuspendOnPaymentFailure(invoice.id);
            console.log(
              `[Payment Monitor] Suspended agreement ${invoice.assignmentId}`
            );
          } else {
            // Send failure notification with retry info
            await sendPaymentFailureNotification(invoice.id, true);
            console.log(
              `[Payment Monitor] Sent retry notification for invoice ${invoice.id}`
            );
          }
        } catch (error) {
          console.error(
            `[Payment Monitor] Error processing invoice ${invoice.id}:`,
            error
          );
        }
      }

      return { processed: failedInvoices.length };
    }
    ```
