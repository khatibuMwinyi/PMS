    ```typescript
    'use server';
    import { prisma } from '@/core/database/client';
    import { nanoid } from 'nanoid';

    /**
     * Write financial audit event (append-only).
     * Every money movement should call this.
     */
    export async function writeFinancialAudit(params: {
      entityType: 'ASSIGNMENT' | 'INVOICE' | 'WALLET' | 'WITHDRAWAL';
      entityId: string;
      action: string;
      payload: any;
      actorId?: string;
    }) {
      await prisma.financialAuditLog.create({
        data: {
          id: nanoid(),
          entityType: params.entityType,
          entityId: params.entityId,
          action: params.action,
          payload: params.payload,
          actorId: params.actorId,
        },
      });
    }

    /**
     * Write general audit event (for non-financial events).
     */
    export async function writeAudit(params: {
      actorId: string;
      entityType: string;
      entityId: string;
      action: 'CREATE' | 'UPDATE' | 'DELETE' | 'STATUS_CHANGE';
      oldValue?: any;
      newValue?: any;
    }) {
      await prisma.auditEvent.create({
        data: {
          id: nanoid(),
          actorId: params.actorId,
          entityType: params.entityType,
          entityId: params.entityId,
          action: params.action,
          oldValue: params.oldValue ?? null,
          newValue: params.newValue ?? null,
        },
      });
    }
    ```

    ## Create src/features/audits/queries.ts
    ```typescript
    'use server';
    import { prisma } from '@/core/database/client';
    import { auth } from '@/core/auth';

    /**
     * Get financial audit logs with filters (admin only).
     */
    export async function getFinancialAuditLogs(params?: {
      entityType?: string;
      entityId?: string;
      startDate?: Date;
      endDate?: Date;
      limit?: number;
    }) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      return await prisma.financialAuditLog.findMany({
        where: {
          entityType: params?.entityType,
          entityId: params?.entityId,
          createdAt: {
            gte: params?.startDate,
            lte: params?.endDate,
          },
        },
        orderBy: { createdAt: 'desc' },
        take: params?.limit ?? 100,
      });
    }

    /**
     * Get general audit events (admin only).
     */
    export async function getAuditEvents(params?: {
      entityType?: string;
      entityId?: string;
      action?: string;
      limit?: number;
    }) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      return await prisma.auditEvent.findMany({
        where: {
          entityType: params?.entityType,
          entityId: params?.entityId,
          action: params?.action,
        },
        orderBy: { createdAt: 'desc' },
        take: params?.limit ?? 100,
      });
    }

    /**
     * Get complete audit trail for a specific entity.
     */
    export async function getEntityAuditTrail(entityType: string, entityId: string) {
      const session = await auth();
      if (!session?.user || !['ADMIN', 'STAFF'].includes(session.user.role)) {
        throw new Error('Unauthorized');
      }

      const [financialLogs, auditEvents] = await Promise.all([
        prisma.financialAuditLog.findMany({
          where: { entityType, entityId },
          orderBy: { createdAt: 'asc' },
        }),
        prisma.auditEvent.findMany({
          where: { entityType, entityId },
          orderBy: { createdAt: 'asc' },
        }),
      ]);

      return { financialLogs, auditEvents };
    }
    ```

    ```tsx
    }) {
    ```

    ```tsx
        <div className="overflow-x-auto">

              <tr className="bg-[var(--surface-card)]">
                <th className="p-2 text-left">Date</th>

                <th className="p-2 text-left">Action</th>
                <th className="p-2 text-left">Details</th>

              {logs.length === 0 ? (
                <tr>
                  <td colSpan={4} className="p-4 text-center text-gray-500">

                  </td>
                </tr>
              ) : (
                logs.map((log) => (
                  <tr key={log.id} className="border-t">
                    <td className="p-2">

                    </td>
                    <td className="p-2">
                      <pre className="text-xs overflow-x-auto">
    ```
