      assignment     Assignment    @relation(fields: [assignmentId], references: [id])

      paidAt         DateTime?
      cancelledAt    DateTime?
      createdAt      DateTime      @default(now())
      updatedAt      DateTime      @updatedAt

      @@index([status, dueDate])
      @@map("invoices")
    }

    ## actions.ts
    ```typescript
    'use server';
    import { prisma } from '@/core/database/client';
    import { nanoid } from 'nanoid';
    import { Decimal } from '@prisma/client/runtime/library';
    import { auth } from '@/core/auth';
    import { GenerateInvoiceSchema, UpdateInvoiceStatusSchema } from './types';

    export async function generateInvoice(input: GenerateInvoiceInput) {
      const session = await auth();
      if (!session?.user) throw new Error('Unauthorized');

      const parsed = GenerateInvoiceSchema.safeParse(input);
      if (!parsed.success) throw new Error('Invalid input');

      const { assignmentId, dueDateOffsetDays } = parsed.data;

      // Get assignment with related data
      const assignment = await prisma.assignment.findUnique({
        where: { id: assignmentId },
        include: {
          serviceType: true,
          property: { include: { owner: true } },
        },
      });

      if (!assignment) throw new Error('Assignment not found');
      if (assignment.status !== 'ACCEPTED') throw new Error('Assignment not accepted');

      // Generate invoice number: INV-YYYYMMDD-XXXX
      const now = new Date();
      const dateStr = now.toISOString().split('T')[0].replace(/-/g, '');
      const random = Math.floor(1000 + Math.random() * 9000);
      const invoiceNumber = `INV-${dateStr}-${random}`;

      // Calculate amount from service type base price
      const amount = new Decimal(assignment.serviceType.basePrice);

      const invoice = await prisma.invoice.create({
        data: {
          id: nanoid(),
          assignmentId,
          invoiceNumber,
          amount,
          dueDate: new Date(Date.now() + dueDateOffsetDays * 24 * 60 * 60 * 1000),
          status: 'PENDING',
        },
      });

      return invoice;
    }

    export async function updateInvoiceStatus(input: UpdateInvoiceStatusInput) {
      const parsed = UpdateInvoiceStatusSchema.safeParse(input);
      if (!parsed.success) throw new Error('Invalid input');

      const { invoiceId, status, retry } = parsed.data;

      const updateData: any = { status };

      if (status === 'PAID') {
        updateData.paidAt = new Date();
      } else if (status === 'CANCELLED') {
        updateData.cancelledAt = new Date();
      }

      if (retry) {
        await prisma.invoice.update({
          where: { id: invoiceId },
          data: {
            retryCount: { increment: 1 },
            status: 'PENDING',
          },
        });
        return;
      }

      return await prisma.invoice.update({
        where: { id: invoiceId },
        data: updateData,
      });
    }
    ```

    'use server';
    import { prisma } from '@/core/database/client';
    import { nanoid } from 'nanoid';
    import { auth } from '@/core/auth';
    import { CreateDisputeSchema } from './types';

    /**
     * Create dispute within 24h of task completion.
     * Freezes funds immediately on creation.
     */
    export async function createDispute(input: CreateDisputeInput) {
      const session = await auth();
      if (!session?.user) throw new Error('Unauthorized');

      const parsed = CreateDisputeSchema.safeParse(input);
      if (!parsed.success) {
        throw new Error(parsed.error.errors[0]?.message ?? 'Invalid input');
      }

      const { taskId, reason, evidenceUrls } = parsed.data;

      // Verify task exists and is completed
      const task = await prisma.task.findUnique({
        where: { id: taskId },
        include: {
          assignment: {
            include: {
              property: { include: { owner: true } },
            },
          },
        },
      });

      if (!task) throw new Error('Task not found');
      if (task.status !== 'COMPLETED') throw new Error('Task not completed');

      // Verify 24h window from task completion
      if (task.checkOutTime) {
        const completionTime = new Date(task.checkOutTime).getTime();
        const now = Date.now();
        const hoursSinceCompletion = (now - completionTime) / (1000 * 60 * 60);

        if (hoursSinceCompletion > 24) {
          throw new Error('Dispute window expired (24h from completion)');
        }
      }

      // Calculate expiresAt (24h from now)
      const expiresAt = new Date(Date.now() + 24 * 60 * 60 * 1000);

      // Create dispute with funds freeze
      const dispute = await prisma.dispute.create({
        data: {
          id: nanoid(),
          taskId,
          reason,
          evidenceUrls,
          fundsFrozen: true,
          expiresAt,
          status: 'OPEN',
        },
      });

      // Freeze funds: mark related wallet transaction as frozen
      // (This will be handled by payment saga compensation)

      return dispute;
    }

    /**
     * Upload evidence to existing dispute
     */
    export async function addDisputeEvidence(disputeId: string, evidenceUrl: string) {
      const session = await auth();
      if (!session?.user) throw new Error('Unauthorized');

      return await prisma.dispute.update({
        where: { id: disputeId },
        data: {
          evidenceUrls: { push: evidenceUrl },
        },
      });
    }
